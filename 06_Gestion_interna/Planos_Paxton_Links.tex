\section*{C\'omo descargar en alta resoluci\'on}
\begin{enumerate}
    \item Abrir cualquiera de los enlaces de arriba en el navegador.
    \item Buscar el bot\'on \textbf{``Download''} (men\'u desplegable) debajo o al lado del visor de la imagen.
    \item Elegir el formato \textbf{TIFF} (m\'axima resoluci\'on, las cotas se leen n\'itidas).
    \item Guardar el archivo en la carpeta \texttt{Drive/Planos}.
\end{enumerate}

\textbf{Para ver otras l\'aminas} (s\'otano, secciones, elevaciones), cambiar el n\'umero
\texttt{sp=} al final de la primera URL: \texttt{?sp=1}, \texttt{?sp=2}, \texttt{?sp=5}, etc.
